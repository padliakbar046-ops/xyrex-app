async function download() {
  const url = document.getElementById('videoUrl').value.trim();
  if (!url) { window.xyrexToast('Isi link video dulu', 'error'); return; }
  const result = document.getElementById('result');
  result.classList.remove('show');

  const btn = document.querySelector('.btn-primary');
  btn.textContent = 'Memproses...';
  btn.disabled = true;

  try {
    const r = await fetch('/api/tools/video?url=' + encodeURIComponent(url), {
      headers: { 'x-auth-token': token }
    });
    const j = await r.json();
    if (!j.ok) {
      window.xyrexToast('Gagal: ' + (j.msg || 'unknown'), 'error');
      btn.textContent = 'Download';
      btn.disabled = false;
      return;
    }

    const player = document.getElementById('videoPlayer');
    player.src = j.url;
    player.load();
    player.onerror = () => {
      window.xyrexToast('Video gagal di-load di player. Coba tombol download.', 'error');
    };
    document.getElementById('videoInfo').textContent = 'Judul: ' + (j.title || '-') + ' | Platform: ' + (j.platform || '-');
    const dl = document.getElementById('dlLink');
    dl.href = j.url;
    dl.setAttribute('download', 'video_' + Date.now() + '.mp4');
    dl.setAttribute('target', '_blank');
    result.classList.add('show');
    btn.textContent = 'Download';
    btn.disabled = false;
  } catch (e) {
    window.xyrexToast('Error: ' + e.message, 'error');
    btn.textContent = 'Download';
    btn.disabled = false;
  }
}
</script>

let currentEmail = '';
let currentPass = '';
let currentToken = '';

async function generateEmail() {
  try {
    const r = await fetch('/api/tools/gmail/create', {
      method: 'POST',
      headers: { 'x-auth-token': token }
    });
    const j = await r.json();
    if (!j.ok) { alert('Gagal: ' + (j.msg || 'unknown')); return; }
    currentEmail = j.email;
    currentPass = j.password;
    currentToken = j.token;
    document.getElementById('emailDisplay').textContent = currentEmail;
    document.getElementById('emailBox').style.display = 'block';
    document.getElementById('status').textContent = 'Menunggu email masuk...';
    document.getElementById('msgList').innerHTML = '';
    setInterval(refreshInbox, 10000);
  } catch (e) {
    alert('Error: ' + e.message);
  }
}

async function refreshInbox() {
  if (!currentToken) return;
  try {
    const r = await fetch('/api/tools/gmail/inbox?token=' + encodeURIComponent(currentToken), {
      headers: { 'x-auth-token': token }
    });
    const j = await r.json();
    if (!j.ok) return;
    const list = document.getElementById('msgList');
    list.innerHTML = '';
    if (!j.messages || j.messages.length === 0) {
      list.innerHTML = '<div style="text-align:center; color:#7a8aaa; padding:16px;">Belum ada email</div>';
      return;
    }
    j.messages.forEach(m => {
      const el = document.createElement('div');
      el.className = 'msg-item';
      el.innerHTML = '<div class="from">' + (m.from || '-') + '</div><div class="subj">' + (m.subject || '-') + '</div><div class="body">' + (m.intro || '-') + '</div>';
      list.appendChild(el);
    });
    document.getElementById('status').textContent = j.messages.length + ' email diterima';
  } catch (e) {}
}

async function loadTrendingAnime() {
  const grid = document.getElementById('animeResult');
  grid.innerHTML = '<div style="grid-column:1/-1; text-align:center; padding:16px; color:#7a8aaa;">Memuat trending...</div>';
  try {
    const r = await fetch('/api/tools/anime/trending', { headers: { 'x-auth-token': token } });
    const j = await r.json();
    if (!j.ok || !j.results || j.results.length === 0) {
      grid.innerHTML = '<div style="grid-column:1/-1; text-align:center; color:#7a8aaa; padding:20px;">Trending tidak tersedia</div>';
      return;
    }
    renderAnime(j.results);
  } catch (e) {
    grid.innerHTML = '<div style="grid-column:1/-1; text-align:center; color:#7a8aaa; padding:20px;">Gagal load trending</div>';
  }
}

function renderAnime(animeList) {
  const grid = document.getElementById('animeResult');
  grid.innerHTML = '';
  animeList.forEach(a => {
    const el = document.createElement('div');
    el.className = 'anime-card';
    el.innerHTML = '<img src="' + (a.image || '') + '" alt="" onerror="this.style.display=\'none\'"><div class="ac-title">' + a.title + '</div>';
    el.onclick = () => watchAnime(a.url);
    grid.appendChild(el);
  });
}

async function searchAnime() {
  const q = document.getElementById('searchInput').value.trim();
  if (!q) { alert('Masukkan judul anime'); return; }
  try {
    const r = await fetch('/api/tools/anime/search?q=' + encodeURIComponent(q), {
      headers: { 'x-auth-token': token }
    });
    const j = await r.json();
    if (!j.ok) { alert('Gagal: ' + (j.msg || 'unknown')); return; }
    const grid = document.getElementById('animeResult');
    grid.innerHTML = '';
    if (!j.results || j.results.length === 0) {
      grid.innerHTML = '<div style="grid-column:1/-1; text-align:center; color:#7a8aaa; padding:20px;">Anime tidak ditemukan</div>';
      return;
    }
    renderAnime(j.results);
  } catch (e) {
    alert('Error: ' + e.message);
  }
}
