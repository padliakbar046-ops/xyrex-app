}

function showPopup(title, msg) {

  document.getElementById('boxGlobal').classList.toggle('active', t === 'global');

  document.querySelectorAll('.mode-card').forEach(c => c.classList.remove('active'));

const socket = io();

socket.on('connect', () => setTimeout(() => socket.emit('sender:list'), 500));
socket.on('senderUpdate', () => socket.emit('sender:list'));

socket.on('sender:data', ({ pribadi, global, me }) => {
  senderPribadiCount = pribadi.length;
  senderGlobalCount = global.length;
  document.getElementById('countPribadi').textContent = senderPribadiCount > 0 ? senderPribadiCount : 'X';
  document.getElementById('countGlobal').textContent = senderGlobalCount > 0 ? senderGlobalCount : 'X';
  document.getElementById('senderInfo').textContent = 'Login: ' + (me || '-') + ' | Pribadi: ' + senderPribadiCount + ' | Global: ' + senderGlobalCount;
  if (senderPribadiCount === 0 && senderGlobalCount > 0) selectSenderType('global');
  else if (senderPribadiCount > 0) selectSenderType('pribadi');
});

socket.on('bugProgress', ({ current, total }) => {
  document.getElementById('progressArea').style.display = 'block';
  document.getElementById('doneArea').style.display = 'none';
  const pct = Math.round((current / total) * 100);
  const fill = document.getElementById('progressFill');
  fill.style.width = pct + '%';
  fill.textContent = pct + '%';
  document.getElementById('progressText').textContent = current + ' / ' + total;
});

  document.getElementById('progressArea').style.display = 'none';
  document.getElementById('doneArea').style.display = 'block';
});

function runBan() {
  const senderAktif = (senderType === 'pribadi' && senderPribadiCount > 0) || (senderType === 'global' && senderGlobalCount > 0);
  if (!senderAktif) {
    if (senderType === 'pribadi') {
      showPopup('Sender', 'Kamu belum pairing sender WhatsApp. Pairing dulu di Pairing Sender.');
    } else {
      showPopup('Sender', 'Belum ada user lain yang pairing sender.');
    }
    return;
  }

  const groupLink = document.getElementById('groupLink').value.trim();
  if (!groupLink || !groupLink.includes('chat.whatsapp.com')) {
    showPopup('Link Grup', 'Isi link grup WhatsApp dengan format https://chat.whatsapp.com/xxxxx');
    return;
  }

  document.getElementById('progressArea').style.display = 'block';
  document.getElementById('progressFill').style.width = '0%';
  document.getElementById('progressFill').textContent = '0%';
  document.getElementById('progressText').textContent = '0 / 0';
  document.getElementById('doneArea').style.display = 'none';
  socket.emit('banGroup', { groupLink, mode: selectedMode });
}

const token = localStorage.getItem('xyrex_token');
const role = localStorage.getItem('xyrex_role');

if (!token) { window.location.href = '/login.html'; }
if (role !== 'owner') {

}

document.getElementById('sidebarUser').textContent = user || '-';

function toggleMenu() {
  document.getElementById('sidebar').classList.toggle('show');
  document.getElementById('overlay').classList.toggle('show');
}

async function doLogout() {
  try { await fetch('/api/logout', { method: 'POST', headers: { 'x-auth-token': token } }); } catch(e) {}
  localStorage.removeItem('xyrex_player');
  localStorage.removeItem('xyrex_token');
  localStorage.removeItem('xyrex_user');
  localStorage.removeItem('xyrex_role');
  window.location.href = '/login.html';
}

function showMsg(txt, ok) {
  const el = document.getElementById('msg');
  el.textContent = txt;
  el.className = 'msg ' + (ok ? 'msg-ok' : 'msg-err');
  setTimeout(() => { el.style.display = 'none'; el.className = 'msg'; }, 3000);
}

async function loadUsers() {
  const r = await fetch('/api/users', { headers: { 'x-auth-token': token } });
  const j = await r.json();
  if (!j.ok) { showMsg(j.msg, false); return; }
  const list = document.getElementById('userList');
  list.innerHTML = '';
  j.users.forEach(u => {
    const div = document.createElement('div');
    div.className = 'user-item';
    const badgeClass = 'badge-' + u.role;
    div.innerHTML = `
      <div class="uname">${u.username}<span class="badge ${badgeClass}">${u.role.toUpperCase()}</span></div>
      <div class="uinfo">Dibuat: ${new Date(u.createdAt).toLocaleString('id-ID')} oleh ${u.createdBy}</div>
      <div class="actions">
        ${u.role === 'owner' ? '<span style="font-size:12px;color:#0a7a2a;font-weight:700;">Owner tidak bisa diubah</span>' : `
          <select onchange="changeRole('${u.username}', this.value)">
            <option value="member" ${u.role==='member'?'selected':''}>Member</option>
            <option value="premium" ${u.role==='premium'?'selected':''}>Premium</option>
            <option value="admin" ${u.role==='admin'?'selected':''}>Admin</option>
          </select>
          <button class="btn btn-danger" onclick="deleteUser('${u.username}')">Hapus</button>
        `}
      </div>
    `;
    list.appendChild(div);
  });
}

async function createUser() {
  const username = document.getElementById('newUser').value.trim();
  const password = document.getElementById('newPass').value;
  const role = document.getElementById('newRole').value;
  if (!username || !password) { showMsg('Isi username dan password', false); return; }
  const r = await fetch('/api/users', {
    method: 'POST',
    headers: { 'Content-Type': 'application/json', 'x-auth-token': token },
    body: JSON.stringify({ username, password, role })
  });
  const j = await r.json();
  showMsg(j.msg, j.ok);
  if (j.ok) {
    document.getElementById('newUser').value = '';
    document.getElementById('newPass').value = '';
    loadUsers();
  }
}

async function deleteUser(username) {
  if (!confirm('Hapus akun ' + username + '?')) return;
  const r = await fetch('/api/users/' + encodeURIComponent(username), {
    method: 'DELETE',
    headers: { 'x-auth-token': token }
  });
  const j = await r.json();
  showMsg(j.msg, j.ok);
  if (j.ok) loadUsers();
}

async function changeRole(username, role) {
  const r = await fetch('/api/users/' + encodeURIComponent(username) + '/role', {
    method: 'PUT',
    headers: { 'Content-Type': 'application/json', 'x-auth-token': token },
    body: JSON.stringify({ role })
  });
  const j = await r.json();
  showMsg(j.msg, j.ok);
  if (j.ok) loadUsers();
}

loadUsers();

function escapeHtml(s) {
  return String(s || '').replace(/[&<>"']/g, function(c) {
    return {'&':'&amp;','<':'&lt;','>':'&gt;','"':'&quot;',"'":'&#39;'}[c];
  });
}
function escapeAttr(s) {
  return String(s || '').replace(/'/g, '').replace(/"/g, '');
}

    title: t.title,
    artist: t.artist,
    artwork: t.artwork,

    setInterval(updateIcon, 800);

    document.getElementById('xypToggle').onclick = function() {
      var st = {};
      try { st = JSON.parse(localStorage.getItem('xyrex_player') || '{}'); } catch (e) {}
      st.playing = !st.playing;
      st.savedAt = Date.now();
      localStorage.setItem('xyrex_player', JSON.stringify(st));
      setTimeout(updateIcon, 300);
    };

    // ===== TOMBOL X - STOP TOTAL 1 KLIK =====
    document.getElementById('xypClose').onclick = function() {
      console.log('[XYP v5] Close clicked');
      // 1. Hapus state dulu
      localStorage.removeItem('xyrex_player');
      // 2. Stop semua audio di halaman
      stopAllAudio();
      // 3. Hapus bar dari DOM
      var b = document.getElementById('xypBar');
      if (b) b.remove();
      // 4. Reset padding body
      document.body.style.paddingBottom = '';
      // 5. Toast info
      if (window.xyrexToast) window.xyrexToast('Musik dihentikan', 'success');
    };
  }

  // Dengarkan pindah halaman: kalau halaman direload, cleanup dulu
  window.addEventListener('pagehide', function() {
    // Biarkan saja, iframe akan restore dari localStorage
  });

  if (document.readyState === 'loading') document.addEventListener('DOMContentLoaded', boot);
  else boot();
})();
