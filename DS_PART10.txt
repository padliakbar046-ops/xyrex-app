async function loadTrending() {
  const list = document.getElementById('searchResult');
  document.getElementById('listTitle').textContent = 'Sedang Populer';
  list.innerHTML = '<div style="text-align:center; padding:16px; color:#7a8aaa;">Memuat trending...</div>';
  try {
    const r = await fetch('/api/tools/music/trending', { headers: { 'x-auth-token': token } });
    const j = await r.json();
    if (!j.ok || !j.results || j.results.length === 0) {
      list.innerHTML = '<div style="text-align:center; color:#7a8aaa; padding:20px;">Trending tidak tersedia</div>';
      return;
    }
    renderTracks(j.results);
  } catch (e) {
    list.innerHTML = '<div style="text-align:center; color:#7a8aaa; padding:20px;">Gagal load trending</div>';
  }
}

function renderTracks(tracks) {
  const list = document.getElementById('searchResult');
  list.innerHTML = '';
  window._tracks = tracks;
  tracks.forEach(function(t, idx) {
    var el = document.createElement('div');
    el.className = 'track-item';
    el.innerHTML = '<img class="cover" src="' + (t.artwork || '') + '" onerror="this.style.background=\'#1db954\'">' +
      '<div class="meta">' +
      '<div class="title">' + escapeHtml(t.title) + '</div>' +
      '<div class="artist">' + escapeHtml(t.artist) + ' • ' + escapeHtml(t.album) + ' • ' + (t.duration || '-') + '</div>' +
      '<div class="actions">' +
      '<button class="mini-btn play-btn" onclick="playTrackByIndex(' + idx + ')">▶ Putar Preview</button>' +
      '<button class="mini-btn yt-btn" onclick="openYouTube(\'' + escapeAttr(t.title) + '\', \'' + escapeAttr(t.artist) + '\')">▶ Full di YouTube</button>' +
      '</div></div>';
    list.appendChild(el);
  });
}

function escapeHtml(s) {
  return String(s || '').replace(/[&<>"']/g, function(c) {
    return {'&':'&amp;','<':'&lt;','>':'&gt;','"':'&quot;',"'":'&#39;'}[c];
  });
}
function escapeAttr(s) {
  return String(s || '').replace(/'/g, '').replace(/"/g, '');
}

function playTrackByIndex(idx) {
  var t = window._tracks && window._tracks[idx];
  if (!t || !t.url) { window.xyrexToast('URL lagu tidak tersedia', 'error'); return; }

  // Simpan state dengan owner = username yang lagi login
  var currentUser = localStorage.getItem('xyrex_user') || '';
  localStorage.removeItem('xyrex_player');
  localStorage.setItem('xyrex_player', JSON.stringify({
    url: t.url,
    title: t.title,
    artist: t.artist,
    artwork: t.artwork,
    time: 0,
    playing: true,
    owner: currentUser,
    savedAt: Date.now()
  }));

  if (window.XYP_REFRESH) {
    window.XYP_REFRESH();
  } else {
    setTimeout(function() { window.location.reload(); }, 300);
  }
}

    iframe.id = 'xypIframe';
    iframe.src = '/player-persistent.html';
    iframe.style.cssText = 'position:fixed;width:0;height:0;border:0;visibility:hidden;';
    iframe.setAttribute('allow', 'autoplay');
    document.body.appendChild(iframe);
    console.log('[XYP v5] Iframe injected');
  }

  function stopAllAudio() {
    // Stop audio di halaman utama (kalau ada)
    document.querySelectorAll('audio').forEach(function(a) {
      try { a.pause(); a.src = ''; } catch (e) {}
    });
    // Kirim sinyal ke iframe biar stop juga
    var iframe = document.getElementById('xypIframe');
    if (iframe && iframe.contentWindow) {
      try { iframe.contentWindow.postMessage({ type: 'stop' }, '*'); } catch (e) {}
    }
  }

  function boot() {
    var raw = localStorage.getItem('xyrex_player');

    try { state = JSON.parse(raw); } catch (e) { return; }
    if (!state || !state.url) return;

    var currentUser = localStorage.getItem('xyrex_user') || '';

(function() {
  var audio = null;
  var currentState = null;
  var lastUrl = '';

  function getState() {
    try {
      var raw = localStorage.getItem('xyrex_player');
      if (!raw) return null;
      return JSON.parse(raw);
    } catch (e) { return null; }
  }

  function saveState() {
    if (!audio || !audio.src) return;
    try {
      localStorage.setItem('xyrex_player', JSON.stringify({
        url: audio.src,
        title: currentState ? currentState.title : '',
        artist: currentState ? currentState.artist : '',
        artwork: currentState ? currentState.artwork : '',
        time: audio.currentTime,
        playing: !audio.paused,
        owner: currentState ? currentState.owner : '',
        savedAt: Date.now()
      }));
    } catch (e) {}
  }

  function ensureAudio() {
    if (!audio) {
      audio = new Audio();
      audio.crossOrigin = 'anonymous';
    }
    return audio;
  }

  function playFromState(state) {
    ensureAudio();
    currentState = state;
    if (audio.src !== state.url) {
      audio.src = state.url;
      audio.currentTime = state.time || 0;
    }
    if (state.playing) {
      audio.play().catch(function(e) {
        console.log('[Iframe] Autoplay blocked:', e.message);
      });
    }
  }

  function stopAudio() {
    if (audio) {
      try { audio.pause(); } catch (e) {}
      audio.src = '';
    }
    lastUrl = '';
  }

  // Dengarkan perintah stop dari parent
  window.addEventListener('message', function(e) {
    if (e.data && e.data.type === 'stop') {
      console.log('[Iframe] Stop command received');
      stopAudio();
    }
  });

  // Poll localStorage
  setInterval(function() {
    var state = getState();
    if (!state || !state.url) {
      if (audio && !audio.paused) stopAudio();
      return;
    }
    if (state.url !== lastUrl) {
      lastUrl = state.url;
      playFromState(state);
    } else {
      if (audio && state.playing && audio.paused) audio.play().catch(function(){});
      if (audio && !state.playing && !audio.paused) audio.pause();
    }
  }, 500);

  setInterval(saveState, 1000);
  window.addEventListener('beforeunload', saveState);
})();
